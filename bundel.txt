    $sources = '';
    foreach (File::allFiles(resource_path('views')) as $f) {
        if (str_contains($f->getFilename(), 'styleguide')) {
            $sources .= $f->getContents();
        }
    }
    expect($sources)->not->toBe('', 'no styleguide view found under resources/views');

    $missing = [];
    foreach (File::files(resource_path('views/components/ui')) as $file) {
        $name = str_replace('.blade.php', '', $file->getFilename());
        if (str_starts_with($name, '_')) {
            continue;
        }
        if (! str_contains($sources, "<x-ui.{$name}") && ! str_contains($sources, "x-ui.{$name} ")) {
            $missing[] = $name;
        }
    }
    expect($missing)->toBeEmpty();
});

    $xp = DesignSystem::xpath(styleguide());
    $ids = [];
    foreach ($xp->query('//*[@id]') as $el) {
        $ids[] = $el->getAttribute('id');
    }
    $dupes = array_keys(array_filter(array_count_values($ids), fn ($n) => $n > 1));
    expect($dupes)->toBeEmpty();
});

it('has one h1 and a heading order that never skips levels', function () {
    $xp = DesignSystem::xpath(styleguide());
    expect($xp->query('//h1')->length)->toBe(1);
    $prev = 0;
    $skips = [];
    foreach ($xp->query('//h1|//h2|//h3|//h4|//h5|//h6') as $h) {
        $level = (int) substr($h->nodeName, 1);
        if ($prev && $level > $prev + 1) {
            $skips[] = "{$h->nodeName} after h{$prev}";
        }
        $prev = $level;
    }
    expect($skips)->toBeEmpty();
});

    $xp = DesignSystem::xpath(styleguide());
    $problems = [];

    foreach ($xp->query('//*[@data-modal-open]') as $trigger) {
        $target = $trigger->getAttribute('data-modal-open');
        if ($xp->query('//dialog[@id="'.$target.'"]')->length !== 1) {
            $problems[] = "modal trigger points to missing dialog #{$target}";
        }
    }
    foreach ($xp->query('//dialog') as $d) {
        $ref = $d->getAttribute('aria-labelledby');
        if ($ref === '' || $xp->query('//*[@id="'.$ref.'"]')->length !== 1) {
            $problems[] = 'dialog without valid aria-labelledby';
        }
    }
    foreach ($xp->query('//*[@role="tablist"]') as $list) {
        $selected = 0;
        foreach ($xp->query('.//*[@role="tab"]', $list) as $tab) {
            $panel = $tab->getAttribute('aria-controls');
            if ($xp->query('//*[@role="tabpanel" and @id="'.$panel.'"]')->length !== 1) {
                $problems[] = "tab controls missing panel #{$panel}";
            }
            if ($tab->getAttribute('aria-selected') === 'true') {
                $selected++;
                if ($tab->getAttribute('tabindex') !== '0') {
                    $problems[] = 'selected tab must have tabindex=0';
                }
            } elseif ($tab->getAttribute('tabindex') !== '-1') {
                $problems[] = 'unselected tab must have tabindex=-1';
            }
        }
        if ($selected !== 1) {
            $problems[] = "tablist has {$selected} selected tabs";
        }
    }
    foreach ($xp->query('//*[@role="tabpanel"]') as $panel) {
        $ref = $panel->getAttribute('aria-labelledby');
        if ($xp->query('//*[@role="tab" and @id="'.$ref.'"]')->length !== 1) {
            $problems[] = 'tabpanel without matching tab';
        }
    }
    expect($problems)->toBe([]);
});

<?php

declare(strict_types=1);

namespace Tests\Support;

use DOMDocument;
use DOMXPath;
use Illuminate\Support\Facades\File;

/**
 * Helpers for design-system conformance tests (DS-01..DS-07).
 * Adjust the constants below if your paths differ.
 */
final class DesignSystem
{
    public const TOKENS = 'resources/css/tokens.css';

    public const PUBLIC_FONTS = 'public/fonts';

    /** @return list<array{rel:string,content:string}> */
    public static function files(array $dirs, array $suffixes): array
    {
        $out = [];
        foreach ($dirs as $dir) {
            $abs = base_path($dir);
            if (! is_dir($abs)) {
                continue;
            }
            foreach (File::allFiles($abs) as $file) {
                foreach ($suffixes as $suffix) {
                    if (str_ends_with($file->getFilename(), $suffix)) {
                        $out[] = [
                            'rel' => ltrim(str_replace(base_path(), '', $file->getPathname()), '/'),
                            'content' => $file->getContents(),
                        ];
                        break;
                    }
                }
            }
        }

        return $out;
    }

    /** @return list<array{rel:string,content:string}> */
    public static function views(): array
    {
        return self::files(['resources/views'], ['.blade.php']);
    }

    /** CSS files except tokens.css (the only place raw colours may live). */
    public static function cssExceptTokens(): array
    {
        return array_values(array_filter(
            self::files(['resources/css'], ['.css']),
            fn (array $f) => $f['rel'] !== self::TOKENS,
        ));
    }

    public static function scripts(): array
    {
        return self::files(['resources/js'], ['.js', '.ts']);
    }

    /** @return array<string,string> custom property => value (as written) */
    public static function tokens(): array
    {
        $css = (string) file_get_contents(base_path(self::TOKENS));
        $css = preg_replace('#/\*.*?\*/#s', '', $css) ?? $css;
        preg_match_all('/(--[a-z0-9-]+)\s*:\s*([^;{}]+);/i', $css, $m, PREG_SET_ORDER);
        $out = [];
        foreach ($m as $match) {
            $out[$match[1]] = trim($match[2]);
        }

        return $out;
    }

    public static function luminance(string $hex): float
    {
        $hex = ltrim($hex, '#');
        if (strlen($hex) === 3) {
            $hex = $hex[0].$hex[0].$hex[1].$hex[1].$hex[2].$hex[2];
        }
        [$r, $g, $b] = array_map(fn ($c) => hexdec($c) / 255, str_split($hex, 2));
        $f = fn (float $c) => $c <= 0.03928 ? $c / 12.92 : (($c + 0.055) / 1.055) ** 2.4;

        return 0.2126 * $f($r) + 0.7152 * $f($g) + 0.0722 * $f($b);
    }

    public static function contrast(string $fg, string $bg): float
    {
        $a = self::luminance($fg);
        $b = self::luminance($bg);
        [$hi, $lo] = $a >= $b ? [$a, $b] : [$b, $a];

        return ($hi + 0.05) / ($lo + 0.05);
    }

    public static function xpath(string $html): DOMXPath
    {
        $dom = new DOMDocument;
        libxml_use_internal_errors(true);
        $dom->loadHTML('<?xml encoding="utf-8" ?>'.$html);
        libxml_clear_errors();

        return new DOMXPath($dom);
    }
}

<?php

declare(strict_types=1);

namespace App\Http\Requests\Editor;

use App\Domain\Invitation\EventTypes\EventTypeDefinition;
use App\Domain\Invitation\EventTypes\EventTypeRegistry;
use App\Domain\Invitation\Models\Invitation;
use Illuminate\Foundation\Http\FormRequest;

/**
 * One request per editor step. Rules come from the event type definition, so adding a field in
 * config/event-types.php or EventTypeDefinition changes validation everywhere at once.
 * The route must bind the invitation as {invitation}.
 */
abstract class EditorStepRequest extends FormRequest
{
    /** One of EventTypeDefinition::STEPS. */
    abstract protected function step(): string;

    public function invitation(): Invitation
    {
        $invitation = $this->route('invitation');

        return $invitation instanceof Invitation ? $invitation : Invitation::query()->findOrFail($invitation);
    }

    public function definition(): EventTypeDefinition
    {

<?php

declare(strict_types=1);

namespace App\Domain\Billing;

use App\Domain\Billing\Models\Plan;
use App\Domain\Invitation\Models\Invitation;
use App\Models\User;
use Carbon\CarbonImmutable;
use Carbon\CarbonInterface;
use InvalidArgumentException;

/** DOM-10: the single place that answers "may this invitation/user do X?". No plan checks anywhere else. */
final class Entitlements
{
    public function __construct(private readonly SubscriptionResolver $resolver) {}

    /** The plan in force: the active subscription's plan, else the free plan. */
    public function planFor(Invitation|User $subject, ?CarbonInterface $at = null): Plan
    {
        $sub = $subject instanceof Invitation
            ? $this->resolver->forInvitation($subject, $at)
            : $this->resolver->forUser($subject, $at);

        return $sub?->plan ?? Plan::free();
    }

    /** True when a non-free subscription is active at the given moment. */
    public function hasActivePlan(Invitation|User $subject, ?CarbonInterface $at = null): bool
    {
        $sub = $subject instanceof Invitation
            ? $this->resolver->forInvitation($subject, $at)
            : $this->resolver->forUser($subject, $at);

        return $sub !== null && ! $sub->plan->isFree();
    }

    public function can(Invitation|User $subject, string $ability, ?CarbonInterface $at = null): bool
    {
        if (! in_array($ability, (array) config('entitlements.abilities'), true)) {
            throw new InvalidArgumentException("Unknown ability [{$ability}].");
        }

        return (bool) ($this->planFor($subject, $at)->features[$ability] ?? false);
    }

    /** null = no limit decided/applied. */
    public function limit(Invitation|User $subject, string $key, ?CarbonInterface $at = null): ?int
    {
        if (! in_array($key, (array) config('entitlements.limits'), true)) {
            throw new InvalidArgumentException("Unknown limit [{$key}].");
        }
        $value = $this->planFor($subject, $at)->limits[$key] ?? null;

        return $value === null ? null : (int) $value;
    }

    /** When a publish made at $from should expire: end of the active subscription, else free-plan duration, else never. */
    public function expiresAtFor(Invitation $invitation, ?CarbonInterface $from = null): ?CarbonImmutable
    {
        $from = CarbonImmutable::instance($from ?? now());
        $sub = $this->resolver->forInvitation($invitation, $from);
        if ($sub?->ends_at !== null) {
            return CarbonImmutable::instance($sub->ends_at);
        }
        $days = ($sub?->plan ?? Plan::free())->duration_days;

        return $days === null ? null : $from->addDays($days);
    }
}

<?php

namespace App\Console\Commands;

use Illuminate\Console\Command;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Storage;
use Symfony\Component\Console\Attribute\AsCommand;

#[AsCommand(name: 'backup:restore')]
class RestoreDatabaseBackup extends Command
{
    /**
     * The name and signature of the console command.
     *
     * @var string
     */
    protected $signature = 'backup:restore
                            {backupName? : The backup name to restore}
                            {--disk= : The disk where the backup is stored}
                            {--force : Skip confirmation prompt}';

    /**
     * The console command description.
     *
     * @var string
     */
    protected $description = 'Restore database from backup';

    /**
     * Execute the console command.
     */
    public function handle(): int
    {
        $disk = $this->option('disk') ?: config('backup.disk', 's3');
        $backupName = $this->argument('backupName') ?: $this->selectBackup($disk);

        if (! $backupName) {
            $this->error('No backup selected');

            return self::FAILURE;
        }

        // Confirm restore
        if (! $this->option('force')) {
            if (! $this->confirm("Are you sure you want to restore backup '{$backupName}'? This will overwrite the current database.")) {
                $this->info('Restore cancelled.');

                return self::FAILURE;
            }
        }

        try {
            $this->info("Starting restore from backup: {$backupName}");

            // Verify manifest exists
            if (! Storage::disk($disk)->exists("backups/{$backupName}/manifest.json")) {
                $this->error("Manifest file not found for backup: {$backupName}");

                return self::FAILURE;
            }

            // Verify backup.sql exists
            if (! Storage::disk($disk)->exists("backups/{$backupName}/backup.sql")) {
                $this->error("Backup SQL file not found for backup: {$backupName}");

                return self::FAILURE;
            }

            // Read manifest for verification
            $manifest = json_decode((string) Storage::disk($disk)->get("backups/{$backupName}/manifest.json"), true);
            $this->info("Manifest loaded for database: {$manifest['database']}");

            // Restore database
            $this->info('Restoring database...');
            $this->restoreDatabase($disk, $backupName);

            // Verify restore
            $this->info('Verifying restore...');
            if ($this->verifyRestore($manifest)) {
                $this->info('Restore completed successfully!');
                $this->info("Backup manifest: {$backupName}/manifest.json");
                $this->info("Backup SQL: {$backupName}/backup.sql");

                return self::SUCCESS;
            }

            $this->error('Restore verification failed - row counts do not match');

            return self::FAILURE;
        } catch (\Exception $e) {
            $this->error("Restore failed: {$e->getMessage()}");

            return self::FAILURE;
        }
    }

    /**
     * Select backup from list of available backups.
     */
    private function selectBackup(string $disk): ?string
    {
        $backups = Storage::disk($disk)->directories('backups');

        if (empty($backups)) {
            $this->error('No backups found');

            return null;
        }

        // Sort by name (timestamp) descending
        rsort($backups);

        $backupNames = array_map('basename', $backups);

        return $this->choice('Select backup to restore', $backupNames);
    }

    /**
     * Restore database from SQL file.
     */
    private function restoreDatabase(string $disk, string $backupName): void
    {
        $sqlContent = (string) Storage::disk($disk)->get("backups/{$backupName}/backup.sql");

        // Split SQL into individual statements
        $statements = array_filter(array_map('trim', explode(';', $sqlContent)));

        // Execute each statement
        foreach ($statements as $statement) {
            if (trim($statement) !== '' && $statement !== '--') {
                try {
                    DB::statement($statement);
                } catch (\Exception $e) {
                    // Skip if statement already exists (for idempotent operations)
                    if (! str_contains($e->getMessage(), 'already exists')) {
                        throw $e;
                    }
                }
            }
        }
    }

    /**
     * Verify restore by comparing row counts with manifest.
     */
    private function verifyRestore(array $manifest): bool
    {
        $allMatch = true;

        foreach ($manifest['tables'] as $table => $expectedCount) {
            if (! DB::getSchemaBuilder()->hasTable($table)) {
                $this->error("Table '{$table}' does not exist after restore");
                $allMatch = false;

                continue;
            }

            $actualCount = DB::table($table)->count();

            if ($actualCount !== $expectedCount) {
                $this->error("Table '{$table}': expected {$expectedCount} rows, got {$actualCount} rows");
                $allMatch = false;
            }
        }

        return $allMatch;
    }
}

<?php

namespace App\Jobs;

use Illuminate\Bus\Queueable;
use Illuminate\Contracts\Queue\ShouldQueue;
use Illuminate\Foundation\Bus\Dispatchable;
use Illuminate\Queue\InteractsWithQueue;
use Illuminate\Queue\SerializesModels;
use Illuminate\Support\Facades\Storage;
use InvalidArgumentException;

class CleanupOldBackups implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable, SerializesModels;

    /**
     * Create a new job instance.
     */
    public function __construct(
        public readonly int $minRetentionDays = 7,
        public readonly int $maxRetentionDays = 30,
        public readonly string $disk = 'backups',
    ) {}

    /**
     * Execute the job.
     */
    public function handle(): void
    {
        $this->validateRetentionConfig();
        $this->cleanupOldBackups();
    }

    /**
     * Validate retention configuration.
     */
    private function validateRetentionConfig(): void
    {
        if ($this->minRetentionDays < 0) {
            throw new InvalidArgumentException('Minimum retention days must be >= 0');
        }

        if ($this->maxRetentionDays < $this->minRetentionDays) {
            throw new InvalidArgumentException("Maximum retention days ({$this->maxRetentionDays}) must be >= minimum retention days ({$this->minRetentionDays})");
        }
    }

    /**
     * Cleanup backups older than max retention days.
     */
    private function cleanupOldBackups(): void
    {
        $backupPath = 'backups';

        if (! Storage::disk($this->disk)->exists($backupPath)) {
            return;
        }

        $directories = Storage::disk($this->disk)->directories($backupPath);

        foreach ($directories as $directory) {
            $timestamp = basename($directory);

            if (! $this->isValidBackupTimestamp($timestamp)) {
                continue;
            }

            $backupDate = \DateTime::createFromFormat('Y-m-d_His', $timestamp);

            if (! $backupDate) {
                continue;
            }

            $ageInDays = $backupDate->diff(now())->days;

            // Delete backups older than max retention days
            if ($ageInDays > $this->maxRetentionDays) {
                $this->deleteBackupDirectory($directory);
            }
        }
    }

    /**
     * Check if timestamp matches backup format.
     */
    private function isValidBackupTimestamp(string $timestamp): bool
    {
        return (bool) \DateTime::createFromFormat('Y-m-d_His', $timestamp);
    }

    /**
     * Delete backup directory and all its contents.
     */
    private function deleteBackupDirectory(string $directory): void
    {
        try {
            Storage::disk($this->disk)->deleteDirectory($directory);
        } catch (\Exception $e) {
            report($e);
        }
    }

    /**
     * Get the number of seconds the job can run.
     */
    public function timeout(): int
    {
        return 600; // 10 minutes maximum
    }
}

use App\Http\Middleware\AddSurfaceHeader;
use App\Http\Middleware\SecurityHeaders;
use Illuminate\Foundation\Application;
use Illuminate\Foundation\Configuration\Exceptions;
use Illuminate\Foundation\Configuration\Middleware;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Log;
use Illuminate\Support\Facades\Redis;
use Illuminate\Support\Facades\Route;
use Sentry\Laravel\Integration;

return Application::configure(basePath: dirname(__DIR__))
    ->withRouting(
        commands: __DIR__.'/../routes/console.php',
        health: '/up',
        then: function () {
            // Subdomain surfaces are registered BEFORE the root domain surface so that
            // Laravel's router matches app.momenikah.test and undangan.momenikah.test
            // before it evaluates the root momenikah.test domain.
            $surfaces = [
                'app' => [
                    'domain' => config('surfaces.app.domain', 'app.momenikah.test'),
                    'file' => base_path('routes/app.php'),
                ],
                'public' => [
                    'domain' => config('surfaces.public.domain', 'undangan.momenikah.test'),
                    'file' => base_path('routes/public.php'),
                ],
                // Root domain registered last — must come after all subdomain entries.
                'marketing' => [
                    'domain' => config('surfaces.marketing.domain', 'momenikah.test'),
                    'file' => base_path('routes/marketing.php'),
                ],
            ];

            foreach ($surfaces as $key => $config) {
                try {
                    if (! file_exists($config['file'])) {
                        throw new RuntimeException("Route file not found: {$config['file']}");
                    }

                    Route::domain($config['domain'])
                        ->middleware(['web', AddSurfaceHeader::class.':'.$key])
                        ->group($config['file']);
                } catch (Throwable $e) {
                    // Per Requirement 5.11: log the failing route file name and halt
                    // initialization so the application does not start in a broken state.
                    Log::critical("Failed to load route file for surface [{$key}]: {$e->getMessage()}", [
                        'surface' => $key,
                        'file' => $config['file'],
                        'exception' => $e,
                    ]);

                    throw new RuntimeException(
                        "Application initialization halted: could not load route file for surface [{$key}]. "
                        ."File: {$config['file']}. Reason: {$e->getMessage()}",
                        previous: $e
                    );
                }
            }

            // Health check endpoint — domain-agnostic, accessible from any surface.
            // Registered after surface routes so surface fallback does not shadow it.
            Route::get('/health', function () {
                $status = 'healthy';
                $checks = [];
                $httpCode = 200;

                try {
                    DB::connection()->getPdo();
                    $checks['database'] = [
                        'status' => 'connected',
                        'connection' => config('database.default'),
                    ];
                } catch (Throwable $e) {
                    $status = 'unhealthy';
                    $httpCode = 503;
                    $checks['database'] = [
                        'status' => 'disconnected',
                        'error' => $e->getMessage(),
                    ];
                }

                try {
                    Redis::ping();
                    $checks['redis'] = [
                        'status' => 'connected',
                    ];
                } catch (Throwable $e) {
                    $status = 'unhealthy';
                    $httpCode = 503;
                    $checks['redis'] = [
                        'status' => 'disconnected',
                        'error' => $e->getMessage(),
                    ];
                }

                return response()->json([
                    'status' => $status,
                    'timestamp' => now()->toISOString(),
                    'checks' => $checks,
                ], $httpCode);
            })->middleware('web')->name('health.check');

            // Fallback: any request reaching here did not match a configured surface domain.
            // Returns HTTP 404 per Requirement 5.5.
            Route::fallback(function () {
                abort(404, 'Unrecognized surface domain.');
            });
        },
    )
    ->withMiddleware(function (Middleware $middleware): void {
        // Register security headers middleware globally for all web routes
        // This ensures all routes across all surfaces (marketing, app, public) receive security headers
        $middleware->prepend(SecurityHeaders::class);

        $middleware->alias([
            'surface' => AddSurfaceHeader::class,
        ]);
    })
    ->withExceptions(function (Exceptions $exceptions): void {
        Integration::handles($exceptions);
    })->create();

# Third-Party Licenses

**Last Updated:** 2024-12-23

This document provides an overview of third-party libraries, frameworks, fonts, and JavaScript packages used in this Laravel 13 application, along with their respective licenses.

## Maintenance

This section documents the process for maintaining the third-party dependency list.

### Adding New Dependencies

When adding a new dependency, follow these steps:

1. **Install the package** using the appropriate command:

   - For Composer (PHP packages):
     ```bash
     composer require vendor/package-name
     ```

   - For npm (JavaScript packages):
     ```bash
     npm install package-name
     ```

2. **Update THIRD_PARTY.md**:
   - Add the new package to the appropriate section (Composer Packages or npm Packages)
   - Include the package name, version, and license information
   - Add a brief note about usage if applicable
   - Update the **Last Updated** date at the top of the document to today's date (ISO 8601 format: YYYY-MM-DD)

3. **Commit changes** in the same commit as the dependency installation:
   ```bash
   git add composer.json composer.lock package.json package-lock.json THIRD_PARTY.md
   git commit -m "Add new dependency and update THIRD_PARTY.md"
   ```

**Example - Adding a new Composer package:**
```bash
# Install the package
composer require spatie/laravel-model-status

# Update THIRD_PARTY.md (add spatie/laravel-model-status to Composer Packages section)
# Update the Last Updated date

# Commit all changes
git add composer.json composer.lock THIRD_PARTY.md
git commit -m "Add spatie/laravel-model-status package"
```

### Removing Dependencies

When removing a dependency, follow these steps:

1. **Remove the package** using the appropriate command:

   - For Composer:
     ```bash
     composer remove vendor/package-name
     ```

   - For npm:
     ```bash
     npm uninstall package-name
     ```

2. **Update THIRD_PARTY.md**:
   - Remove the package entry from the appropriate section
   - Reorder packages alphabetically if needed for consistency
   - Update the **Last Updated** date at the top of the document to today's date (ISO 8601 format: YYYY-MM-DD)

3. **Commit changes** in the same commit as the dependency removal:
   ```bash
   git add composer.json composer.lock package.json package-lock.json THIRD_PARTY.md
   git commit -m "Remove dependency and update THIRD_PARTY.md"
   ```

**Example - Removing a Composer package:**
```bash
# Remove the package
composer remove vendor/package-name

# Update THIRD_PARTY.md (remove package entry)
# Update the Last Updated date

# Commit all changes
git add composer.json composer.lock THIRD_PARTY.md
git commit -m "Remove vendor/package-name"
```

### Updating Last-Updated Date

- **Always update the Last Updated date** when making any changes to the dependency list (adding, removing, or modifying entries)
- Use ISO 8601 format: `YYYY-MM-DD` (e.g., `2024-12-23`)
- Update the date at the top of the document, below the main heading

### Future Automation

Consider implementing a pre-commit hook to automatically update the Last Updated date when changes are made to dependency files. This could be implemented using a simple script that:

1. Detects changes to `composer.lock` or `package-lock.json`
2. Updates the Last Updated date in `THIRD_PARTY.md`
3. Automatically stages the updated file

This would ensure the date is always current without manual intervention.
